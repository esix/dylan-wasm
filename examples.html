Dylan source (in hello/hello.dylan)
define function fact (n :: <integer>) => (r :: <integer>)
  if (n <= 1) 1 else n * fact(n - 1) end
end;

define function hanoi
    (n :: <integer>, from :: <string>, to :: <string>, via :: <string>) => ()
  if (n = 1)
    format-out("  disk 1: %s -> %s\n", from, to);
  else
    hanoi(n - 1, from, via, to);
    format-out("  disk %d: %s -> %s\n", n, from, to);
    hanoi(n - 1, via, to, from);
  end if;
end function;

towers-of-hanoi — native output (the live demo above also includes a smaller 3-disk hanoi)

Classic recursion: move 5 disks in 2⁵−1 = 31 moves.

Dylan source
define method hanoi
    (from :: <tower>, to :: <tower>, with :: <tower>,
     #key count :: <integer> = from.height)
  if (count >= 1)
    hanoi(from, with, to, count: count - 1);
    move-disk(from, to);
    hanoi(with, to, from, count: count - 1)
  end
end;
