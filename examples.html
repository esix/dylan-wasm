    (v :: <sequence>) => (sorted-v :: <sequence>)

          let t = v[m]; v[m] := v[n]; v[n] := t
        end;

towers-of-hanoi

Classic recursion: move 5 disks in 2⁵−1 = 31 moves.

Dylan source
define method hanoi
    (from :: <tower>, to :: <tower>, with :: <tower>,
     #key count :: <integer> = from.height)
  if (count >= 1)
    hanoi(from, with, to, count: count - 1);
    move-disk(from, to);
    hanoi(with, to, from, count: count - 1)
  end
end;
